 backend:
    build: ./backend
    container_name: meeting_room_backend
    ports:
      - "5000:5000"
    env_file:
      - ./backend/.env
    environment:
      - DB_HOST=postgres
      - DB_PORT=5432
      - REDIS_HOST=redis
      - DB_PASSWORD=postgres
      - NODE_ENV=production
    depends_on:
      postgres:
        condition: service_healthy
      redis:
        condition: service_healthy


  frontend:
    build:
      context: ./frontend
      args:
        - VITE_API_URL=http://localhost:5000
    container_name: meeting_room_frontend
    ports:
      - "3000:80"
    depends_on:
      - backend
